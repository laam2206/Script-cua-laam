-- XỬ LÝ SỰ KIỆN NÚT BẤM
GetKeyBtn.MouseButton1Click:Connect(function()
    if setclipboard then
        setclipboard(LinkGetKey)
        GetKeyBtn.Text = "ĐÃ COPY LINK!"
        task.wait(2)
        GetKeyBtn.Text = "GET KEY"
    else
        TextBox.Text = LinkGetKey
        GetKeyBtn.Text = "LINK Ở Ô NHẬP!"
    end
end)

-- Hàm xử lý khi check key thành công
local function XuLyThanhCong()
    CheckBtn.Text = "KEY ĐÚNG! ĐANG LƯU..."
    CheckBtn.BackgroundColor3 = Color3.fromRGB(76, 175, 80)
    
    if writefile then
        writefile(FileName, CorrectKey)
    end
    
    task.wait(1)
    ScreenGui:Destroy() -- Tắt menu nhập key
    ChayScriptChinh()   -- Chạy script chính
end

CheckBtn.MouseButton1Click:Connect(function()
    if TextBox.Text == CorrectKey then
        XuLyThanhCong()
    else
        CheckBtn.Text = "SAI KEY RỒI!"
        CheckBtn.BackgroundColor3 = Color3.fromRGB(244, 67, 54)
        task.wait(2)
        CheckBtn.Text = "CHECK KEY"
        CheckBtn.BackgroundColor3 = Color3.fromRGB(218, 165, 32)
    end
end)

-- ====================================================================
-- LOGIC KIỂM TRA TỰ ĐỘNG KHI MỞ MENU (AUTO-FILL & AUTO-CHECK)
-- ====================================================================
task.spawn(function()
    task.wait(0.5) -- Chờ giao diện hiện lên mượt mà
    if readfile and isfile and isfile(FileName) then
        local SavedKey = readfile(FileName)
        if SavedKey == CorrectKey then
            TextBox.Text = CorrectKey -- Tự điền key vào ô nhập
            task.wait(0.5) -- Giữ lại nửa giây cho người dùng thấy key đã điền
            XuLyThanhCong() -- Tự chạy hiệu ứng thành công và vào script chính
        end
    end
end)
