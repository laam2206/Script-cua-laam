--// 🛠️ HWID
local function getHWID()
    local hwid = nil
    pcall(function()
        if gethwid then hwid = gethwid()
        elseif syn and syn.get_hwid then hwid = syn.get_hwid()
        end
    end)
    if not hwid then hwid = tostring(LocalPlayer.UserId) .. "_" .. LocalPlayer.Name end
    return hwid
end

--// 🌐 GỌI API
local function callAPI(action, key)
    local url = API_URL .. "?action=" .. action
        .. "&key=" .. (key or "")
        .. "&hwid=" .. getHWID()
        .. "&secret=" .. SECRET
    
    local ok, res = pcall(function()
        return game:HttpGet(url, true)
    end)
    
    if not ok then
        return { success = false, message = "❌ Không kết nối server!" }
    end
    
    local ok2, data = pcall(function()
        return HttpService:JSONDecode(res)
    end)
    
    if ok2 and data then return data end
    return { success = false, message = "❌ Lỗi server!" }
end

--// 💾 LƯU KEY
local SAVE_FILE = "LamHub_Key.json"

local function saveKey(k)
    pcall(function()
        if writefile then
            writefile(SAVE_FILE, HttpService:JSONEncode({ key = k }))
        end
    end)
end

local function loadSaved()
    local ok, d = pcall(function()
        if readfile and isfile and isfile(SAVE_FILE) then
            return HttpService:JSONDecode(readfile(SAVE_FILE))
        end
    end)
    if ok and d and d.key then return d.key end
    return nil
end

local function clearSaved()
    pcall(function()
        if delfile and isfile and isfile(SAVE_FILE) then delfile(SAVE_FILE) end
    end)
end

local keyInfo = { expTime = 0, isValid = false }

local function showStatus(text, color)
    StatusLabel.Text = text
    StatusLabel.TextColor3 = color
end

local function formatTime(seconds)
    if seconds <= 0 then return "Đã hết hạn" end
    local days = math.floor(seconds / 86400)
    local hours = math.floor((seconds % 86400) / 3600)
    local mins = math.floor((seconds % 3600) / 60)
    if days > 0 then
        return days .. " ngày " .. hours .. " giờ"
    elseif hours > 0 then
        return hours .. " giờ " .. mins .. " phút"
    else
        return mins .. " phút"
    end
end

local function updateTimeDisplay()
    if not keyInfo.isValid then
        TimeBox.Visible = false
        return
    end
    TimeBox.Visible = true
    if keyInfo.expTime == 0 then
        TimeLabel.Text = "⏰ Còn lại: ♾️ Vĩnh viễn"
        TimeLabel.TextColor3 = Color3.fromRGB(255, 200, 87)
    else
        local remain = keyInfo.expTime - os.time()
        if remain <= 0 then
            TimeLabel.Text = "❌ Key đã hết hạn!"
            TimeLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
        else
            TimeLabel.Text = "⏰ Còn lại: " .. formatTime(remain)
            TimeLabel.TextColor3 = Color3.fromRGB(255, 180, 180)
        end
    end
end

local function parseTimeFromMessage(msg)
    if msg:find("Vinh vien") or msg:find("Vinh Viễn") then
        keyInfo.expTime = 0
        keyInfo.isValid = true
        return
    end
    local days = tonumber(msg:match("(%d+) ngay"))
    local hours = tonumber(msg:match("(%d+) gio")) or 0
    local mins = tonumber(msg:match("(%d+) phut")) or 0
    if days then
        keyInfo.expTime = os.time() + days * 86400 + hours * 3600
    elseif hours > 0 then
        keyInfo.expTime = os.time() + hours * 3600 + mins * 60
    else
        keyInfo.expTime = os.time() + mins * 60
    end
    keyInfo.isValid = true
end

SubmitBtn.MouseButton1Click:Connect(function()
    local k = TextBox.Text
    if k == "" then
        showStatus("⚠️ Vui lòng nhập key!", Color3.fromRGB(255, 200, 60))
        return
    end
    
    showStatus("⏳ Đang kiểm tra key...", Color3.fromRGB(120, 200, 255))
    
    local r = callAPI("validate", k)
    
    if r.success then
        showStatus("✅ " .. r.message, Color3.fromRGB(80, 255, 120))
        parseTimeFromMessage(r.message)
        updateTimeDisplay()
        saveKey(k)
        task.wait(1.5)
        
        -- 🌟 CHẠY MENU CHÍNH
        loadMainMenu()
    else
        showStatus("❌ " .. r.message, Color3.fromRGB(255, 80, 80))
        keyInfo.isValid = false
        TimeBox.Visible = false
        
        local orig = KeyFrame.Position
        for i = 1, 3 do
            TweenService:Create(KeyFrame, TweenInfo.new(0.05), {
                Position = orig + UDim2.new(0, 10, 0, 0)
            }):Play()
            task.wait(0.05)
            TweenService:Create(KeyFrame, TweenInfo.new(0.05), {
                Position = orig - UDim2.new(0, 10, 0, 0)
            }):Play()
            task.wait(0.05)
        end
        TweenService:Create(KeyFrame, TweenInfo.new(0.05), {Position = orig}):Play()
    end
end)
