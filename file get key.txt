--// 🛠️ HWID
local function getHWID()
    local hwid = nil
    pcall(function()
        if gethwid then hwid = gethwid()
        elseif syn and syn.get_hwid then hwid = syn.get_hwid()
        end
    end)
    if not hwid then hwid = tostring(player.UserId) .. "_" .. player.Name end
    return hwid
end

--// 🌐 API
local function callAPI(action, key)
    local url = API_URL .. "?action=" .. action
        .. "&key=" .. (key or "")
        .. "&hwid=" .. getHWID()
        .. "&secret=" .. SECRET
    
    local ok, res = pcall(function()
        return game:HttpGet(url, true)
    end)
    
    if not ok then
        return { success = false, message = "❌ Không kết nối server!" }
    end
    
    local ok2, data = pcall(function()
        return HttpService:JSONDecode(res)
    end)
    
    if ok2 and data then return data end
    return { success = false, message = "❌ Lỗi server!" }
end

--// 💾 SAVE KEY
local function saveKey(k)
    pcall(function()
        if writefile then
            writefile("LamHub_Key.json", HttpService:JSONEncode({key = k}))
        end
    end)
end

local function loadSaved()
    local ok, d = pcall(function()
        if readfile and isfile and isfile("LamHub_Key.json") then
            return HttpService:JSONDecode(readfile("LamHub_Key.json"))
        end
    end)
    if ok and d and d.key then return d.key end
    return nil
end

local function clearSaved()
    pcall(function()
        if delfile and isfile and isfile("LamHub_Key.json") then
            delfile("LamHub_Key.json")
        end
    end)
end

local keyInfo = { expTime = 0, isValid = false }

local function showStatus(text, color)
    Status.Text = text
    Status.TextColor3 = color
end

local function formatTime(sec)
    if sec <= 0 then return "Đã hết hạn" end
    local d = math.floor(sec / 86400)
    local h = math.floor((sec % 86400) / 3600)
    local m = math.floor((sec % 3600) / 60)
    if d > 0 then return d .. " ngày " .. h .. " giờ"
    elseif h > 0 then return h .. " giờ " .. m .. " phút"
    else return m .. " phút" end
end

local function updateTime()
    if not keyInfo.isValid then TimeBox.Visible = false return end
    TimeBox.Visible = true
    if keyInfo.expTime == 0 then
        TimeLabel.Text = "⏰ ♾️ Vĩnh viễn"
        TimeLabel.TextColor3 = Color3.fromRGB(255, 200, 87)
    else
        local remain = keyInfo.expTime - os.time()
        if remain <= 0 then
            TimeLabel.Text = "❌ Đã hết hạn"
            TimeLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
        else
            TimeLabel.Text = "⏰ " .. formatTime(remain)
            TimeLabel.TextColor3 = Color3.fromRGB(255, 180, 180)
        end
    end
end

local function parseTime(msg)
    if msg:find("Vinh vien") then
        keyInfo.expTime = 0
        keyInfo.isValid = true
        return
    end
    local d = tonumber(msg:match("(%d+) ngay"))
    local h = tonumber(msg:match("(%d+) gio")) or 0
    local m = tonumber(msg:match("(%d+) phut")) or 0
    if d then
        keyInfo.expTime = os.time() + d * 86400 + h * 3600
    elseif h > 0 then
        keyInfo.expTime = os.time() + h * 3600 + m * 60
    else
        keyInfo.expTime = os.time() + m * 60
    end
    keyInfo.isValid = true
end

    end)
end

--// 🎯 SUBMIT
SubmitBtn.MouseButton1Click:Connect(function()
    local k = TextBox.Text
    if k == "" then
        showStatus("⚠️ Nhập key!", Color3.fromRGB(255, 200, 60))
        return
    end
    
    showStatus("⏳ Đang check...", Color3.fromRGB(120, 200, 255))
    
    local r = callAPI("validate", k)
    
    if r.success then
        showStatus("✅ " .. r.message, Color3.fromRGB(80, 255, 120))
        parseTime(r.message)
        updateTime()
        saveKey(k)
        task.wait(1.2)
        loadMenu()
    else
        showStatus("❌ " .. r.message, Color3.fromRGB(255, 80, 80))
        keyInfo.isValid = false
        TimeBox.Visible = false
        local orig = KeyFrame.Position
        for i = 1, 3 do
            TweenService:Create(KeyFrame, TweenInfo.new(0.05), {Position = orig + UDim2.new(0, 8, 0, 0)}):Play()
            task.wait(0.05)
            TweenService:Create(KeyFrame, TweenInfo.new(0.05), {Position = orig - UDim2.new(0, 8, 0, 0)}):Play()
            task.wait(0.05)
        end
        TweenService:Create(KeyFrame, TweenInfo.new(0.05), {Position = orig}):Play()
    end
end)
