-- ============================================
-- LẤY HWID (CỐ ĐỊNH)
-- ============================================
local function GetHWID()
    local userId = tostring(LocalPlayer.UserId)
    local executorName = "unknown"
    if identifyexecutor then
        local ok, name = pcall(identifyexecutor)
        if ok and name then executorName = name end
    end
    return "USER_" .. userId .. "_" .. executorName
end

-- ============================================
-- LƯU KEY 3 LỚP
-- ============================================
local KEY_FILES = {"LaamHub_Key.txt", "LaamHub_SavedKey.txt", "laam_key.dat"}

local function SaveKey(key)
    getgenv().LaamHub_Key = key
    if writefile then
        for _, file in ipairs(KEY_FILES) do
            pcall(writefile, file, key)
        end
    end
end

local function ReadSavedKey()
    if getgenv().LaamHub_Key and getgenv().LaamHub_Key ~= "" then
        return getgenv().LaamHub_Key
    end
    if isfile and readfile then
        for _, file in ipairs(KEY_FILES) do
            local ok, exists = pcall(isfile, file)
            if ok and exists then
                local ok2, content = pcall(readfile, file)
                if ok2 and content and content ~= "" then
                    getgenv().LaamHub_Key = content
                    return content
                end
            end
        end
    end
    return nil
end

local function DeleteSavedKey()
    getgenv().LaamHub_Key = nil
    if delfile then
        for _, file in ipairs(KEY_FILES) do
            pcall(delfile, file)
        end
    end
end

-- ============================================
-- XÁC THỰC KEY QUA API
-- ============================================
local function VerifyKeyOnline(inputKey)
    local hwid = GetHWID()
    local payload = HttpService:JSONEncode({
        key = inputKey,
        hwid = hwid,
        username = LocalPlayer.Name,
    })
    
    local responseBody = nil
    
    if request then
        local ok, response = pcall(function()
            return request({
                Url = CONFIG.API_URL,
                Method = "POST",
                Headers = {["Content-Type"] = "application/json"},
                Body = payload,
            })
        end)
        if ok and response then responseBody = response.Body or response end
    end
    
    if not responseBody and syn and syn.request then
        local ok, response = pcall(function()
            return syn.request({
                Url = CONFIG.API_URL,
                Method = "POST",
                Headers = {["Content-Type"] = "application/json"},
                Body = payload,
            })
        end)
        if ok and response then responseBody = response.Body or response end
    end
    
    if not responseBody and http_request then
        local ok, response = pcall(function()
            return http_request({
                Url = CONFIG.API_URL,
                Method = "POST",
                Headers = {["Content-Type"] = "application/json"},
                Body = payload,
            })
        end)
        if ok and response then responseBody = response.Body or response end
    end
    
    if not responseBody then
        local ok, response = pcall(function()
            return HttpService:PostAsync(CONFIG.API_URL, payload, Enum.HttpContentType.ApplicationJson)
        end)
        if ok then responseBody = response end
    end
    
    if not responseBody then
        return false, "Không kết nối được server!"
    end
    
    local ok, data = pcall(HttpService.JSONDecode, HttpService, responseBody)
    if not ok then
        return false, "Server trả về lỗi!"
    end
    
    return data.valid, data.message
end

local function TryOpenMenu()
    local ok, err = pcall(function()
        loadstring(game:HttpGet(CONFIG.SCRIPT_URL))()
    end)
    
    if not ok then
        warn("Lỗi load script chính: " .. tostring(err))
        return false
    end
    
    return true
end
