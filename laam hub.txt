local allTabs = {}
local currentTab = nil

function MakeTab(config)
    local tabData = {
        name = config.Name,
        items = {},
        index = #allTabs + 1
    }
    table.insert(allTabs, tabData)

    local Btn = Instance.new("TextButton")
    Btn.Size = UDim2.new(1, 0, 0, 26)
    Btn.BackgroundColor3 = Color3.fromRGB(30, 30, 34)
    Btn.Text = ""
    Btn.BorderSizePixel = 0
    Btn.AutoButtonColor = false
    Btn.Parent = SidebarScroll

    local BtnCorner = Instance.new("UICorner")
    BtnCorner.CornerRadius = UDim.new(0, 5)
    BtnCorner.Parent = Btn

    local Label = Instance.new("TextLabel")
    Label.Size = UDim2.new(1, -24, 1, 0)
    Label.Position = UDim2.new(0, 8, 0, 0)
    Label.BackgroundTransparency = 1
    Label.Text = config.Name
    Label.TextColor3 = Color3.fromRGB(200, 200, 205)
    Label.TextSize = 10
    Label.Font = Enum.Font.Gotham
    Label.TextXAlignment = Enum.TextXAlignment.Left
    Label.TextTruncate = Enum.TextTruncate.AtEnd
    Label.Parent = Btn

    local Arrow = Instance.new("TextLabel")
    Arrow.Size = UDim2.new(0, 16, 1, 0)
    Arrow.Position = UDim2.new(1, -18, 0, 0)
    Arrow.BackgroundTransparency = 1
    Arrow.Text = "›"
    Arrow.TextColor3 = Color3.fromRGB(120, 120, 125)
    Arrow.TextSize = 13
    Arrow.Font = Enum.Font.GothamBold
    Arrow.Parent = Btn

    tabData.button = Btn
    tabData.label = Label

    Btn.MouseEnter:Connect(function()
        if currentTab ~= tabData then
            TweenService:Create(Btn, TweenInfo.new(0.15), {
                BackgroundColor3 = Color3.fromRGB(40, 40, 45)
            }):Play()
        end
    end)
    Btn.MouseLeave:Connect(function()
        if currentTab ~= tabData then
            TweenService:Create(Btn, TweenInfo.new(0.15), {
                BackgroundColor3 = Color3.fromRGB(30, 30, 34)
            }):Play()
        end
    end)

    local function renderItems()
        for _, child in pairs(ContentScroll:GetChildren()) do
            if child:IsA("TextButton") then
                child:Destroy()
            end
        end
        for _, item in ipairs(tabData.items) do
            local ItemBtn = Instance.new("TextButton")
            ItemBtn.Size = UDim2.new(1, 0, 0, 28)
            ItemBtn.BackgroundColor3 = Color3.fromRGB(32, 32, 36)
            ItemBtn.Text = ""
            ItemBtn.BorderSizePixel = 0
            ItemBtn.AutoButtonColor = false
            ItemBtn.Parent = ContentScroll

            local ItemCorner = Instance.new("UICorner")
            ItemCorner.CornerRadius = UDim.new(0, 5)
            ItemCorner.Parent = ItemBtn

            local ItemLabel = Instance.new("TextLabel")
            ItemLabel.Size = UDim2.new(1, -16, 1, 0)
            ItemLabel.Position = UDim2.new(0, 10, 0, 0)
            ItemLabel.BackgroundTransparency = 1
            ItemLabel.Text = item.Name
            ItemLabel.TextColor3 = Color3.fromRGB(220, 220, 225)
            ItemLabel.TextSize = 11
            ItemLabel.Font = Enum.Font.Gotham
            ItemLabel.TextXAlignment = Enum.TextXAlignment.Left
            ItemLabel.TextTruncate = Enum.TextTruncate.AtEnd
            ItemLabel.Parent = ItemBtn

            ItemBtn.MouseEnter:Connect(function()
                TweenService:Create(ItemBtn, TweenInfo.new(0.15), {
                    BackgroundColor3 = Color3.fromRGB(50, 50, 58)
                }):Play()
            end)
            ItemBtn.MouseLeave:Connect(function()
                TweenService:Create(ItemBtn, TweenInfo.new(0.15), {
                    BackgroundColor3 = Color3.fromRGB(32, 32, 36)
                }):Play()
            end)
            ItemBtn.MouseButton1Click:Connect(function()
                if item.Callback then
                    item.Callback()
                end
            end)
        end
    end

    tabData.render = renderItems

    Btn.MouseButton1Click:Connect(function()
        if currentTab then
            TweenService:Create(currentTab.button, TweenInfo.new(0.2), {
                BackgroundColor3 = Color3.fromRGB(30, 30, 34)
            }):Play()
            currentTab.label.TextColor3 = Color3.fromRGB(200, 200, 205)
        end

        currentTab = tabData
        TweenService:Create(Btn, TweenInfo.new(0.2), {
            BackgroundColor3 = Color3.fromRGB(60, 100, 200)
        }):Play()
        Label.TextColor3 = Color3.fromRGB(255, 255, 255)

        ContentTitle.Text = config.Name
        renderItems()
    end)

    if #allTabs == 1 then
        Btn.BackgroundColor3 = Color3.fromRGB(60, 100, 200)
        Label.TextColor3 = Color3.fromRGB(255, 255, 255)
        currentTab = tabData
        ContentTitle.Text = config.Name
    end

    return tabData
end

function AddButton(tabData, config)
    table.insert(tabData.items, {
        Name = config.Name,
        Callback = config.Callback
    })

    if currentTab == tabData then
        tabData.render()
    end
end

local menuOpen = false

LogoFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
    or input.UserInputType == Enum.UserInputType.Touch then
        menuOpen = not menuOpen
        if menuOpen then
            MainFrame.Visible = true
            MainFrame.Size = UDim2.new(0, 480, 0, 0)
            TweenService:Create(MainFrame, TweenInfo.new(0.3, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
                Size = UDim2.new(0, 480, 0, 330)
            }):Play()
            TweenService:Create(LogoStroke, TweenInfo.new(0.2), {
                Color = Color3.fromRGB(80, 180, 80)
            }):Play()
        else
            local tween = TweenService:Create(MainFrame, TweenInfo.new(0.2), {
                Size = UDim2.new(0, 480, 0, 0)
            })
            tween:Play()
            tween.Completed:Connect(function()
                MainFrame.Visible = false
            end)
            TweenService:Create(LogoStroke, TweenInfo.new(0.2), {
                Color = Color3.fromRGB(220, 30, 30)
            }):Play()
        end
    end
end)

LogoFrame.MouseEnter:Connect(function()
    TweenService:Create(LogoFrame, TweenInfo.new(0.15), {
        Size = UDim2.new(0, 63, 0, 63)
    }):Play()
end)
LogoFrame.MouseLeave:Connect(function()
    TweenService:Create(LogoFrame, TweenInfo.new(0.15), {
        Size = UDim2.new(0, 55, 0, 55)

end)

-- Lấy tên game
task.spawn(function()
    local ok, name = pcall(function()
        return game:GetService("MarketplaceService"):GetProductInfo(game.PlaceId).Name
    end)
    if ok and name then
        GameLabel.Text = "🎮 " .. name
    else
        GameLabel.Text = "🎮 " .. tostring(game.PlaceId)
    end
end)

-- FPS (0.5s)
task.spawn(function()
    local frames = 0
    local lastTime = tick()
    RunService.RenderStepped:Connect(function()
        frames = frames + 1
        local now = tick()
        if now - lastTime >= 0.5 then
            local fps = math.floor(frames / (now - lastTime))
            FpsLabel.Text = "📊 FPS: " .. fps
            if fps >= 50 then
                FpsLabel.TextColor3 = Color3.fromRGB(150, 255, 150)
            elseif fps >= 30 then
                FpsLabel.TextColor3 = Color3.fromRGB(255, 255, 150)
            else
                FpsLabel.TextColor3 = Color3.fromRGB(255, 120, 120)
            end
            frames = 0
            lastTime = now
        end
    end)
end)

-- Ping (0.5s)
task.spawn(function()
    while InfoFrame.Parent do
        local ok, ping = pcall(function()
            return Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
        end)
        if ok and ping then
            ping = math.floor(ping)
            PingLabel.Text = "📡 Ping: " .. ping .. " ms"
            if ping < 80 then
                PingLabel.TextColor3 = Color3.fromRGB(150, 255, 150)
            elseif ping < 150 then
                PingLabel.TextColor3 = Color3.fromRGB(255, 255, 150)
            else
                PingLabel.TextColor3 = Color3.fromRGB(255, 120, 120)
            end
        else
            PingLabel.Text = "📡 Ping: N/A"
        end
        task.wait(0.5)
    end
end)

    while InfoFrame.Parent do
        local count = #Players:GetPlayers()
        PlayerLabel.Text = "👥 Players: " .. count .. "/" .. Players.MaxPlayers
        task.wait(2)
    end
end)
